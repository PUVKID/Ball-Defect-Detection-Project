+    "def _rank_auc_perm(x, y, n_perm=2000, seed=0):\n",
+    "    \"\"\"AUC of feature x for the label y (True = NG) from ranks, and a two-sided permutation p-value. NaNs are dropped.\"\"\"\n",

+    "    ok = ~np.isnan(x)\n",

+    "    cnt = 0\n",
